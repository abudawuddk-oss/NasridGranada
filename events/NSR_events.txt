		NOT = { has_country_flag = NSR_fitna_started }
	}
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					country_exists = SAU
					NOT = { has_country_flag = NSR_saudi_reconciled }
					NOT = { has_country_flag = NSR_fitna_started }
				}
				country_event = { id = nsr.10 days = 365 random_days = 180 }
			}
		}
	}
	option = {
		name = nsr.10.a
		ai_chance = { factor = 3 }
		add_war_support = 0.03
		add_power_balance_value = { id = NSR_dual_power_bop value = 0.03 tooltip_side = NSR_bop_ikhwan_side }
		if = {
			limit = { country_exists = SAU }
			SAU = { add_opinion_modifier = { target = ROOT modifier = NSR_extradition_refused } }
		}
	}
	option = {
		name = nsr.10.b
		trigger = {
			has_character = NSR_meqaid_al_duhainah
			NOT = { has_country_flag = NSR_ikhwan_coup_done }
			NOT = { has_country_flag = NSR_p_refuse_the_king }
		}
		ai_chance = { factor = 1 }
		if = {
			limit = { has_character = NSR_meqaid_al_duhainah }
			retire_character = NSR_meqaid_al_duhainah
		}
		add_power_balance_value = { id = NSR_dual_power_bop value = -0.08 tooltip_side = NSR_bop_court_side }
		custom_effect_tooltip = NSR_tt_fitna_p10
		hidden_effect = {
			add_to_variable = { NSR_fitna = 10 }
			NSR_clamp_turmoil = yes
		}
		add_stability = -0.03
		if = {
			limit = { country_exists = SAU }
			SAU = { add_opinion_modifier = { target = ROOT modifier = NSR_ikhwan_handed_over } }
		}
		set_country_flag = NSR_handed_over_meqaid
	}
}
